cmake_minimum_required(VERSION 3.23)
project(DGLAB_NX_NRO LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UI_BACKEND borealis CACHE STRING "borealis or framebuffer")
if(NOT UI_BACKEND MATCHES "^(borealis|framebuffer)$")
    message(FATAL_ERROR "Unknown UI_BACKEND=${UI_BACKEND}")
endif()
set(DGLAB_PORTLIBS "$ENV{DEVKITPRO}/portlibs/switch" CACHE PATH "Official Switch portlibs prefix")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(ARCH -march=armv8-a+crc+crypto -mtune=cortex-a57 -mtp=soft -fPIE)
add_compile_options(${ARCH} -ffunction-sections -fdata-sections)
file(GLOB_RECURSE NRO_C CONFIGURE_DEPENDS source/*.c)
add_executable(DGLAB-NX ${NRO_C})
set_target_properties(DGLAB-NX PROPERTIES SUFFIX ".elf")
target_include_directories(DGLAB-NX PRIVATE include "${ROOT}/common/include" "${ROOT}/third_party")
target_compile_options(DGLAB-NX PRIVATE -Wall -O2)
target_compile_definitions(DGLAB-NX PRIVATE __SWITCH__
    DGLAB_APP_VERSION="${APP_VERSION}" DGLAB_BUILD_STAMP="${BUILD_STAMP}")
target_link_options(DGLAB-NX PRIVATE ${ARCH}
    -Wl,--gc-sections "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/DGLAB-NX.map")
set(NRO_RESOURCES)
if(UI_BACKEND STREQUAL "borealis")
    foreach(dep glfw3 EGL glapi drm_nouveau)
        if(NOT EXISTS "${DGLAB_PORTLIBS}/lib/lib${dep}.a")
            message(FATAL_ERROR "Missing Switch ${dep}: install switch-glfw switch-mesa switch-glm or run make -C nro deps for a local cache (DGLAB_PORTLIBS=${DGLAB_PORTLIBS})")
        endif()
    endforeach()
    if(NOT EXISTS "${DGLAB_PORTLIBS}/include/glm/glm.hpp")
        message(FATAL_ERROR "Missing switch-glm in DGLAB_PORTLIBS")
    endif()
    set(PLATFORM_SWITCH ON)
    set(BOREALIS_USE_OPENGL ON)
    set(BRLS_CUSTOM_PLATFORM ON)
    set(BRLS_UNITY_BUILD OFF)
    # Custom platform owns HID/services; notifications and audio are unused.
    add_compile_definitions(DGLAB_BOREALIS_MINIMAL __SWITCH__ __GLFW__)
    include_directories("${DGLAB_PORTLIBS}/include")
    link_directories("${DGLAB_PORTLIBS}/lib")
    add_subdirectory("${ROOT}/third_party/borealis/library" "borealis")
    target_compile_definitions(borealis PRIVATE STBTT_STATIC)
    target_link_directories(DGLAB-NX PRIVATE "${DGLAB_PORTLIBS}/lib")
    target_sources(DGLAB-NX PRIVATE source/borealis/native.cpp source/borealis/gameplay.cpp source/borealis/about.cpp)
    target_compile_definitions(DGLAB-NX PRIVATE DGLAB_UI_BOREALIS=1)
    target_link_libraries(DGLAB-NX PRIVATE borealis)
    set(NRO_RESOURCES "--romfsdir=${ROOT}/third_party/borealis/resources")
endif()
target_link_libraries(DGLAB-NX PRIVATE nx m)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/nacp-input.txt" CONTENT "${APP_TITLE}\n${APP_AUTHOR}\n${APP_VERSION}\n")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/DGLAB-NX.nacp"
    COMMAND "${NX_NACPTOOL_EXE}" --create "${APP_TITLE}" "${APP_AUTHOR}" "${APP_VERSION}" "DGLAB-NX.nacp"
    DEPENDS "${ROOT}/VERSION" "${CMAKE_CURRENT_SOURCE_DIR}/Makefile" "${CMAKE_CURRENT_BINARY_DIR}/nacp-input.txt"
    VERBATIM)
file(GLOB_RECURSE ASSETS CONFIGURE_DEPENDS "${ROOT}/third_party/borealis/resources/*")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/DGLAB-NX.nro"
    COMMAND "${NX_ELF2NRO_EXE}" "$<TARGET_FILE:DGLAB-NX>" "DGLAB-NX.nro"
        "--icon=${CMAKE_CURRENT_SOURCE_DIR}/DGLAB-NX.jpg" "--nacp=${CMAKE_CURRENT_BINARY_DIR}/DGLAB-NX.nacp" ${NRO_RESOURCES}
    DEPENDS DGLAB-NX "${CMAKE_CURRENT_BINARY_DIR}/DGLAB-NX.nacp" DGLAB-NX.jpg ${ASSETS}
    VERBATIM)
add_custom_target(nro ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/DGLAB-NX.nro")
